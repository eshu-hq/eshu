import (
	"context"
	"database/sql"
	"encoding/json"
	"fmt"
	"os"
	"testing"
	"time"

	storagepostgres "github.com/eshu-hq/eshu/go/internal/storage/postgres"
	"github.com/eshu-hq/eshu/go/internal/testutil/postgresproof"
)

func TestZZB1BlocksScratch(t *testing.T) {
	ctx, db := postgresproof.OpenDisposableBootstrapDatabase(t,
		os.Getenv("ESHU_GROUPED_INVENTORY_PROOF_DSN"), os.Getenv("ESHU_GROUPED_INVENTORY_PROOF_DISPOSABLE"), 20*time.Minute,
		func(ctx context.Context, d *sql.DB) error {
			return storagepostgres.ApplyBootstrap(ctx, storagepostgres.SQLDB{DB: d})
		})
	const scopes, per = 2000, 50
	for _, s := range []string{
		`INSERT INTO ingestion_scopes (scope_id, scope_kind, source_system, source_key, collector_kind, partition_key, observed_at, ingested_at, status, active_generation_id, payload)
SELECT 'scope:bf:' || n, 'synthetic', 'synthetic', 'k' || n, 'synthetic', 'k' || n, NOW(), NOW(), 'active', 'gen:bf:' || n, '{}'::jsonb FROM generate_series(1, $1) AS n`,
		`INSERT INTO scope_generations (generation_id, scope_id, trigger_kind, observed_at, ingested_at, status, activated_at, payload)
SELECT 'gen:bf:' || n, 'scope:bf:' || n, 'synthetic', NOW(), NOW(), 'active', NOW(), '{}'::jsonb FROM generate_series(1, $1) AS n`,
	} {
		if _, err := db.ExecContext(ctx, s, scopes); err != nil {
			t.Fatal(err)
		}
	}
	if _, err := db.ExecContext(ctx, `
INSERT INTO fact_records (fact_id, scope_id, generation_id, fact_kind, stable_fact_key, schema_version, collector_kind, fencing_token, source_confidence, source_system, source_fact_key, observed_at, ingested_at, payload)
SELECT 'reducer_supply_chain_impact_finding:bf:' || n || ':' || k, 'scope:bf:' || n, 'gen:bf:' || n, 'reducer_supply_chain_impact_finding',
       'k' || n || ':' || k, '1.0.0', 'synthetic', 1, 'observed', 'synthetic', 'k' || n || ':' || k, NOW(), NOW(),
       jsonb_build_object('finding_id', 'finding:bf:' || n || ':' || k, 'cve_id', 'CVE-2026-' || k, 'package_id', 'pkg:npm/p' || k,
         'repository_id', 'repository:bf:' || n, 'impact_status', 'affected_exact', 'priority_score', (k * 7) % 100,
         'priority_bucket', 'high', 'cvss_score', 7.5, 'observed_version', '1.0.0', 'detection_profile', 'precise',
         'match_reason', 'npm_semver_affected_range', 'padding', repeat('x', 600))
FROM generate_series(1, $1) AS n, generate_series(1, $2) AS k`, scopes, per); err != nil {
		t.Fatal(err)
	}
	b := storagepostgres.SupplyChainImpactSupportsBackfill{DB: storagepostgres.SQLDB{DB: db}}
	for {
		r, err := b.RunStep(ctx)
		if err != nil {
			t.Fatal(err)
		}
		if r.Complete {
			break
		}
	}
	for _, tbl := range []string{"fact_records", "supply_chain_impact_finding_supports", "ingestion_scopes", "scope_generations"} {
		if _, err := db.ExecContext(ctx, "ANALYZE "+tbl); err != nil {
			t.Fatal(err)
		}
	}
	candidate := supportsCandidateListQuery(t)
	type grant struct {
		name         string
		scopes, repo []string
	}
	mk := func(from, to int) []string {
		var out []string
		for n := from; n <= to; n++ {
			out = append(out, fmt.Sprintf("scope:bf:%d", n))
		}
		return out
	}
	var repos20 []string
	for n := 1; n <= 20; n++ {
		repos20 = append(repos20, fmt.Sprintf("repository:bf:%d", n*50))
	}
	grants := []grant{{"5 scopes", mk(1, 5), nil}, {"80 scopes", mk(1, 80), nil}, {"400 scopes", mk(1, 400), nil}, {"20 repos", nil, repos20}}
	blocks := func(query string, args []any) (int64, float64) {
		var raw []byte
		if err := db.QueryRowContext(ctx, "EXPLAIN (ANALYZE, BUFFERS, FORMAT JSON) "+query, args...).Scan(&raw); err != nil {
			t.Fatal(err)
		}
		var out []struct {
			Plan struct {
				Hit  int64 `json:"Shared Hit Blocks"`
				Read int64 `json:"Shared Read Blocks"`
			} `json:"Plan"`
			Exec float64 `json:"Execution Time"`
		}
		if err := json.Unmarshal(raw, &out); err != nil {
			t.Fatal(err)
		}
		return out[0].Plan.Hit + out[0].Plan.Read, out[0].Exec
	}
	for _, filterName := range []string{"status", "profile precise"} {
		for _, g := range grants {
			filter := FindingFilter{Limit: 50, Status: "affected_exact", AllowedScopeIDs: g.scopes, AllowedRepositoryIDs: g.repo}
			if filterName == "profile precise" {
				filter.DetectionProfile = ProfilePrecise
			}
			args := multiSupportWinnersListArgs(ctx, filter)
			// Two runs each; report the first (the cold-ish one) and the second.
			lb1, lt1 := blocks(ListFindingsQuery, args)
			cb1, ct1 := blocks(candidate, args)
			lb2, lt2 := blocks(ListFindingsQuery, args)
			cb2, ct2 := blocks(candidate, args)
			t.Logf("%-16s %-10s legacy blocks %7d / %7d  %7.1f / %7.1f ms | candidate blocks %6d / %6d  %6.1f / %6.1f ms | blocks ratio %.3f", filterName, g.name, lb1, lb2, lt1, lt2, cb1, cb2, ct1, ct2, float64(cb1)/float64(lb1))
		}
	}
}
