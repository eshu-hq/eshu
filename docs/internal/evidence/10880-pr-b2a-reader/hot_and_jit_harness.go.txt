// noTxDB hides BeginTx so withJITOff takes its plain branch: the "no wrapper" side
// of the 20% measurement, through the same store.
type noTxDB struct{ database *sql.DB }

func (d noTxDB) QueryContext(ctx context.Context, query string, args ...any) (*sql.Rows, error) {
	return d.database.QueryContext(ctx, query, args...)
}

func (d noTxDB) QueryRowContext(ctx context.Context, query string, args ...any) *sql.Row {
	return d.database.QueryRowContext(ctx, query, args...)
}

func timings(fn func() error, n int) (median, p95, max float64, err error) {
	if err = fn(); err != nil {
		return
	}
	ds := make([]float64, 0, n)
	for i := 0; i < n; i++ {
		start := time.Now()
		if err = fn(); err != nil {
			return
		}
		ds = append(ds, float64(time.Since(start).Microseconds())/1000)
	}
	sort.Float64s(ds)
	return ds[len(ds)/2], ds[(len(ds)*95)/100], ds[len(ds)-1], nil
}

// TestZZJITTwentyPercent measures the three V reads with and without the wrapper,
// interleaved, through the real stores.
func TestZZJITTwentyPercent(t *testing.T) {
	dsn := os.Getenv("ESHU_ZZ_DSN")
	if dsn == "" || os.Getenv("ESHU_ZZ_MODE") != "jit" {
		t.Skip("ESHU_ZZ_DSN / ESHU_ZZ_MODE=jit not set")
	}
	database, err := sql.Open("pgx", dsn)
	if err != nil {
		t.Fatal(err)
	}
	database.SetMaxOpenConns(2)
	ctx := context.Background()
	with := PostgresFindingStore{DB: database, ReadFromWinners: true}
	without := PostgresFindingStore{DB: noTxDB{database}, ReadFromWinners: true}
	aggWith := PostgresAggregateStore{DB: database, ReadFromWinners: true}
	aggWithout := PostgresAggregateStore{DB: noTxDB{database}, ReadFromWinners: true}
	listFilter := FindingFilter{DetectionProfile: ProfilePrecise, Status: "affected_exact", Limit: 201}
	countStatus := AggregateFilter{DetectionProfile: ProfilePrecise, Status: "affected_exact"}
	countNone := AggregateFilter{DetectionProfile: ProfilePrecise}
	cases := []struct {
		name         string
		wrapped, raw func() error
	}{
		{"V list status", func() error { _, e := with.ListSupplyChainImpactFindings(ctx, listFilter); return e },
			func() error { _, e := without.ListSupplyChainImpactFindings(ctx, listFilter); return e }},
		{"V count status", func() error { _, e := aggWith.CountSupplyChainImpactFindings(ctx, countStatus); return e },
			func() error { _, e := aggWithout.CountSupplyChainImpactFindings(ctx, countStatus); return e }},
		{"V count no selector", func() error { _, e := aggWith.CountSupplyChainImpactFindings(ctx, countNone); return e },
			func() error { _, e := aggWithout.CountSupplyChainImpactFindings(ctx, countNone); return e }},
	}
	for _, tc := range cases {
		var wrapped, raw []float64
		if tc.wrapped() != nil || tc.raw() != nil {
			t.Fatalf("%s warm-up failed", tc.name)
		}
		for i := 0; i < 25; i++ {
			s := time.Now()
			_ = tc.raw()
			raw = append(raw, float64(time.Since(s).Microseconds())/1000)
			s = time.Now()
			_ = tc.wrapped()
			wrapped = append(wrapped, float64(time.Since(s).Microseconds())/1000)
		}
		sort.Float64s(wrapped)
		sort.Float64s(raw)
		saving := (raw[len(raw)/2] - wrapped[len(wrapped)/2]) / raw[len(raw)/2] * 100
		t.Logf("RESULT %-22s jit on (no wrapper) median %8.1f ms | wrapper (jit off) median %8.1f ms | net saving %5.1f %%  [n=25 each, interleaved]",
			tc.name, raw[len(raw)/2], wrapped[len(wrapped)/2], saving)
	}
}

// TestZZHotKey times the V, image and inventory reads and the legacy read of the
// same filters on the hot corpus.
func TestZZHotKey(t *testing.T) {
	dsn := os.Getenv("ESHU_ZZ_DSN")
	if dsn == "" || os.Getenv("ESHU_ZZ_MODE") != "hot" {
		t.Skip("ESHU_ZZ_DSN / ESHU_ZZ_MODE=hot not set")
	}
	database, err := sql.Open("pgx", dsn)
	if err != nil {
		t.Fatal(err)
	}
	database.SetMaxOpenConns(2)
	ctx := context.Background()
	gated := PostgresFindingStore{DB: database, ReadFromWinners: true}
	legacy := PostgresFindingStore{DB: database, ReadFromWinners: false}
	aggGated := PostgresAggregateStore{DB: database, ReadFromWinners: true}
	aggLegacy := PostgresAggregateStore{DB: database, ReadFromWinners: false}
	filters := map[string]AggregateFilter{
		"package pkg:hot (1.6M supports)":      {DetectionProfile: ProfilePrecise, PackageID: "pkg:hot"},
		"cve CVE-HOT (20% of the table)":       {DetectionProfile: ProfilePrecise, CVEID: "CVE-HOT"},
		"cve CVE-HOT + package pkg:hot":        {DetectionProfile: ProfilePrecise, CVEID: "CVE-HOT", PackageID: "pkg:hot"},
		"package pkg:hot, status":              {DetectionProfile: ProfilePrecise, PackageID: "pkg:hot", Status: "affected_exact"},
		"no selector (precise)":                {DetectionProfile: ProfilePrecise},
	}
	names := make([]string, 0, len(filters))
	for name := range filters {
		names = append(names, name)
	}
	sort.Strings(names)
	runs := 20
	for _, name := range names {
		f := filters[name]
		lf := FindingFilter{DetectionProfile: f.DetectionProfile, PackageID: f.PackageID, CVEID: f.CVEID, Status: f.Status, Limit: 201}
		if lf.Status == "" && lf.PackageID == "" && lf.CVEID == "" {
			lf.Status = "affected_exact"
		}
		type read struct {
			label string
			fn    func() error
			n     int
		}
		for _, r := range []read{
			{"V list", func() error { _, e := gated.ListSupplyChainImpactFindings(ctx, lf); return e }, runs},
			{"V count", func() error { _, e := aggGated.CountSupplyChainImpactFindings(ctx, f); return e }, runs},
			{"V inventory status", func() error { _, e := aggGated.SupplyChainImpactInventory(ctx, f, InventoryByStatus, 50, 0); return e }, runs},
			{"legacy list", func() error { _, e := legacy.ListSupplyChainImpactFindings(ctx, lf); return e }, 3},
			{"legacy count", func() error { _, e := aggLegacy.CountSupplyChainImpactFindings(ctx, f); return e }, 3},
			{"legacy inventory status", func() error { _, e := aggLegacy.SupplyChainImpactInventory(ctx, f, InventoryByStatus, 50, 0); return e }, 3},
		} {
			median, p95, max, e := timings(r.fn, r.n)
			if e != nil {
				t.Logf("RESULT %-40s %-24s ERROR %v", name, r.label, e)
				continue
			}
			t.Logf("RESULT %-40s %-24s median %9.1f p95 %9.1f max %9.1f ms (n=%d)", name, r.label, median, p95, max, r.n)
		}
	}
	_ = fmt.Sprint
}
