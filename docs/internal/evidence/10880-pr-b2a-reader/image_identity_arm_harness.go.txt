func imgOpen(t *testing.T, dsn string, extra map[string]string, idle int) *sql.DB {
	t.Helper()
	u, err := url.Parse(dsn)
	if err != nil {
		t.Fatal(err)
	}
	q := u.Query()
	for k, v := range extra {
		q.Set(k, v)
	}
	u.RawQuery = q.Encode()
	d, err := sql.Open("pgx", u.String())
	if err != nil {
		t.Fatal(err)
	}
	d.SetMaxOpenConns(1)
	d.SetMaxIdleConns(idle)
	return d
}

// TestZZImageIdentityArm times the image explain statement with a non-empty
// identity on the largest seeded image, under custom and generic plans.
func TestZZImageIdentityArm(t *testing.T) {
	dsn := os.Getenv("ESHU_ZZ_DSN")
	if dsn == "" || os.Getenv("ESHU_ZZ_MODE") != "img" {
		t.Skip("ESHU_ZZ_DSN / ESHU_ZZ_MODE=img not set")
	}
	ctx := context.Background()
	admin := imgOpen(t, dsn, nil, 1)
	const image = "registry.example/hot:1"
	var supports int
	if err := admin.QueryRowContext(ctx, `SELECT count(*) FROM supply_chain_impact_finding_supports WHERE image_ref = $1`, image).Scan(&supports); err != nil {
		t.Fatal(err)
	}
	var lowerFact, lowerKey, winnerFact, winnerKey string
	if err := admin.QueryRowContext(ctx, `
SELECT s.fact_id, s.canonical_key FROM supply_chain_impact_finding_supports s
JOIN supply_chain_impact_canonical_winners w ON w.canonical_key = s.canonical_key
WHERE s.image_ref = $1 AND s.fact_id <> w.winner_fact_id LIMIT 1`, image).Scan(&lowerFact, &lowerKey); err != nil {
		t.Fatal(err)
	}
	if err := admin.QueryRowContext(ctx, `
SELECT w.winner_fact_id, w.canonical_key FROM supply_chain_impact_finding_supports s
JOIN supply_chain_impact_canonical_winners w ON w.canonical_key = s.canonical_key
WHERE s.image_ref = $1 AND s.fact_id = w.winner_fact_id LIMIT 1`, image).Scan(&winnerFact, &winnerKey); err != nil {
		t.Fatal(err)
	}
	t.Logf("IMAGE %s supports=%d lowerFact=%s winnerFact=%s", image, supports, lowerFact, winnerFact)
	identities := []struct{ name, id string }{
		{"lower-copy fact id", lowerFact}, {"canonical key", lowerKey}, {"winner fact id", winnerFact},
	}
	for _, id := range identities {
		args := supplyChainImpactExplanationQueryArgs(ctx, packet.ExplanationFilter{FindingID: id.id, ImageRef: image}, nil)
		for _, mode := range []string{"custom plan (fresh connection each run)", "generic plan (force_generic_plan, one connection)"} {
			var d *sql.DB
			if mode[0] == 'c' {
				d = imgOpen(t, dsn, nil, 0)
			} else {
				d = imgOpen(t, dsn, map[string]string{"plan_cache_mode": "force_generic_plan"}, 1)
			}
			run := func() (int, error) {
				rows, err := d.QueryContext(ctx, ExplainFindingFromImageQuery, args...)
				if err != nil {
					return 0, err
				}
				defer func() { _ = rows.Close() }()
				n := 0
				for rows.Next() {
					n++
				}
				return n, rows.Err()
			}
			warmStart := time.Now()
			if _, err := run(); err != nil {
				t.Fatalf("%s %s warm-up: %v", id.name, mode, err)
			}
			t.Logf("RESULT first execution (the warm-up) identity=%-20s %-48s %8.1f ms", id.name, mode, float64(time.Since(warmStart).Microseconds())/1000)
			var ms []float64
			rowsOut := 0
			for i := 0; i < 5; i++ {
				s := time.Now()
				n, err := run()
				if err != nil {
					t.Fatalf("%s %s: %v", id.name, mode, err)
				}
				rowsOut = n
				ms = append(ms, float64(time.Since(s).Microseconds())/1000)
			}
			sort.Float64s(ms)
			var accesses []planAccess
			var err error
			if mode[0] == 'c' {
				accesses, err = explainAccesses(ctx, admin, ExplainFindingFromImageQuery, args)
			} else {
				accesses, err = explainGenericAccesses(ctx, admin, ExplainFindingFromImageQuery, nil, nil)
			}
			ok, reason := false, ""
			if err == nil {
				ok, reason = imageReadUsesTheImageIndex(accesses)
			} else {
				reason = err.Error()
			}
			t.Logf("RESULT identity=%-20s %-48s median %8.1f ms  max %8.1f  rows=%d  plan names image_ref_idx and no Seq Scan: %v %s",
				id.name, mode, ms[2], ms[4], rowsOut, ok, reason)
			_ = d.Close()
		}
	}
}
