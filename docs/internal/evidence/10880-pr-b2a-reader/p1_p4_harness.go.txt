// noTxDB hides BeginTx so withJITOff takes its plain branch: the "no wrapper" side.
type noTxDB struct{ database *sql.DB }

func (d noTxDB) QueryContext(ctx context.Context, query string, args ...any) (*sql.Rows, error) {
	return d.database.QueryContext(ctx, query, args...)
}

func (d noTxDB) QueryRowContext(ctx context.Context, query string, args ...any) *sql.Row {
	return d.database.QueryRowContext(ctx, query, args...)
}

func openLab(t *testing.T, dsn, application string, extra map[string]string) *sql.DB {
	t.Helper()
	u, err := url.Parse(dsn)
	if err != nil {
		t.Fatal(err)
	}
	q := u.Query()
	q.Set("application_name", application)
	for k, v := range extra {
		q.Set(k, v)
	}
	u.RawQuery = q.Encode()
	database, err := sql.Open("pgx", u.String())
	if err != nil {
		t.Fatal(err)
	}
	database.SetMaxOpenConns(1)
	return database
}

func refresh(ctx context.Context, d *sql.DB) {
	_, _ = d.ExecContext(ctx, `UPDATE supply_chain_impact_winners_materialization SET materialized_at = now()`)
}

// TestZZP4SteadyArm is the arbiter's P4: the generic-plan steady arm through the Go
// stores, each arm on its own connection (raw handle used only through noTxDB), at
// least six warm-ups per arm, a generic plan asserted on the raw connection before
// sampling, n=25 interleaved, all three V reads.
func TestZZP4SteadyArm(t *testing.T) {
	dsn := os.Getenv("ESHU_ZZ_DSN")
	if dsn == "" || os.Getenv("ESHU_ZZ_MODE") != "p4" {
		t.Skip("ESHU_ZZ_DSN / ESHU_ZZ_MODE=p4 not set")
	}
	ctx := context.Background()
	rawDB := openLab(t, dsn, "p4_raw", nil)
	wrapDB := openLab(t, dsn, "p4_wrapped", nil)
	admin := openLab(t, dsn, "p4_admin", nil)
	raw := PostgresFindingStore{DB: noTxDB{rawDB}, ReadFromWinners: true}
	wrapped := PostgresFindingStore{DB: wrapDB, ReadFromWinners: true}
	rawAgg := PostgresAggregateStore{DB: noTxDB{rawDB}, ReadFromWinners: true}
	wrapAgg := PostgresAggregateStore{DB: wrapDB, ReadFromWinners: true}
	listFilter := FindingFilter{DetectionProfile: ProfilePrecise, Status: "affected_exact", Limit: 201}
	countStatus := AggregateFilter{DetectionProfile: ProfilePrecise, Status: "affected_exact"}
	countNone := AggregateFilter{DetectionProfile: ProfilePrecise}
	cases := []struct {
		name          string
		rawFn, wrapFn func() error
	}{
		{"V list status", func() error {
			refresh(ctx, admin)
			_, e := raw.ListSupplyChainImpactFindings(ctx, listFilter)
			return e
		},
			func() error {
				refresh(ctx, admin)
				_, e := wrapped.ListSupplyChainImpactFindings(ctx, listFilter)
				return e
			}},
		{"V count status", func() error {
			refresh(ctx, admin)
			_, e := rawAgg.CountSupplyChainImpactFindings(ctx, countStatus)
			return e
		},
			func() error {
				refresh(ctx, admin)
				_, e := wrapAgg.CountSupplyChainImpactFindings(ctx, countStatus)
				return e
			}},
		{"V count no selector", func() error {
			refresh(ctx, admin)
			_, e := rawAgg.CountSupplyChainImpactFindings(ctx, countNone)
			return e
		},
			func() error {
				refresh(ctx, admin)
				_, e := wrapAgg.CountSupplyChainImpactFindings(ctx, countNone)
				return e
			}},
	}
	for _, tc := range cases {
		for i := 0; i < 8; i++ { // warm-ups: custom plans, then the generic plan
			if err := tc.rawFn(); err != nil {
				t.Fatalf("%s raw warm-up: %v", tc.name, err)
			}
			if err := tc.wrapFn(); err != nil {
				t.Fatalf("%s wrapped warm-up: %v", tc.name, err)
			}
		}
		var generic int
		if err := rawDB.QueryRowContext(ctx, `SELECT COALESCE(max(generic_plans), 0) FROM pg_prepared_statements WHERE statement LIKE '%b_candidates AS (%'`).Scan(&generic); err != nil {
			t.Fatalf("read generic_plans: %v", err)
		}
		t.Logf("PLANSTATE %-22s raw connection max(generic_plans) of the V statement (b_candidates) = %d", tc.name, generic)
		if generic < 1 {
			t.Fatalf("%s: no generic plan on the raw connection after 8 warm-ups: the steady arm is not measuring steady state", tc.name)
		}
		var rawMs, wrapMs []float64
		for i := 0; i < 25; i++ {
			s := time.Now()
			_ = tc.rawFn()
			rawMs = append(rawMs, float64(time.Since(s).Microseconds())/1000)
			s = time.Now()
			_ = tc.wrapFn()
			wrapMs = append(wrapMs, float64(time.Since(s).Microseconds())/1000)
		}
		sort.Float64s(rawMs)
		sort.Float64s(wrapMs)
		saving := (rawMs[12] - wrapMs[12]) / rawMs[12] * 100
		t.Logf("RESULT %-22s generic-plan steady state, separate connections: no wrapper median %8.1f ms | wrapper median %8.1f ms | net saving %5.1f %% [n=25 interleaved, 8 warm-ups, generic_plans=%d]",
			tc.name, rawMs[12], wrapMs[12], saving, generic)
	}
}

// TestZZP1Capture runs each V read of the hot package and hot CVE for the P1 plan
// captures; application_name labels the read in the server log (log_line_prefix %a).
func TestZZP1Capture(t *testing.T) {
	dsn := os.Getenv("ESHU_ZZ_DSN")
	if dsn == "" || os.Getenv("ESHU_ZZ_MODE") != "p1" {
		t.Skip("ESHU_ZZ_DSN / ESHU_ZZ_MODE=p1 not set")
	}
	ctx := context.Background()
	admin := openLab(t, dsn, "p1_admin", nil)
	type selector struct{ label, pkg, cve string }
	for _, workMem := range []string{"", "4MB"} {
		extra := map[string]string{}
		suffix := "wm16"
		if workMem != "" {
			extra["work_mem"] = workMem
			suffix = "wm4"
		}
		for _, sel := range []selector{{"axios", "pkg:npm/axios", ""}, {"hotcve", "", "CVE-HOT"}} {
			if workMem != "" && sel.label != "hotcve" {
				continue // the 4 MB capture is one labelled count
			}
			for _, read := range []string{"list", "count", "inventory"} {
				if workMem != "" && read != "count" {
					continue
				}
				app := fmt.Sprintf("p1_%s_%s_%s", sel.label, read, suffix)
				database := openLab(t, dsn, app, extra)
				for run := 0; run < 3; run++ {
					refresh(ctx, admin)
					var err error
					switch read {
					case "list":
						_, err = PostgresFindingStore{DB: database, ReadFromWinners: true}.ListSupplyChainImpactFindings(ctx,
							FindingFilter{DetectionProfile: ProfilePrecise, PackageID: sel.pkg, CVEID: sel.cve, Limit: 201})
					case "count":
						_, err = PostgresAggregateStore{DB: database, ReadFromWinners: true}.CountSupplyChainImpactFindings(ctx,
							AggregateFilter{DetectionProfile: ProfilePrecise, PackageID: sel.pkg, CVEID: sel.cve})
					case "inventory":
						_, err = PostgresAggregateStore{DB: database, ReadFromWinners: true}.SupplyChainImpactInventory(ctx,
							AggregateFilter{DetectionProfile: ProfilePrecise, PackageID: sel.pkg, CVEID: sel.cve}, InventoryByStatus, 50, 0)
					}
					if err != nil {
						t.Fatalf("%s: %v", app, err)
					}
				}
				_ = database.Close()
				t.Logf("CAPTURED %s", app)
			}
		}
	}
}


// TestZZP4Capture logs the plans of the steady-state raw executions of each V read
// (auto_explain on): the raw connection runs 8 warm-ups, then 3 captured runs, each
// read under its own application_name.
func TestZZP4Capture(t *testing.T) {
	dsn := os.Getenv("ESHU_ZZ_DSN")
	if dsn == "" || os.Getenv("ESHU_ZZ_MODE") != "p4cap" {
		t.Skip("ESHU_ZZ_DSN / ESHU_ZZ_MODE=p4cap not set")
	}
	ctx := context.Background()
	admin := openLab(t, dsn, "p4cap_admin", nil)
	for _, read := range []string{"list_status", "count_status", "count_none"} {
		database := openLab(t, dsn, "p4cap_"+read, nil)
		raw := PostgresFindingStore{DB: noTxDB{database}, ReadFromWinners: true}
		rawAgg := PostgresAggregateStore{DB: noTxDB{database}, ReadFromWinners: true}
		for i := 0; i < 11; i++ {
			refresh(ctx, admin)
			var err error
			switch read {
			case "list_status":
				_, err = raw.ListSupplyChainImpactFindings(ctx, FindingFilter{DetectionProfile: ProfilePrecise, Status: "affected_exact", Limit: 201})
			case "count_status":
				_, err = rawAgg.CountSupplyChainImpactFindings(ctx, AggregateFilter{DetectionProfile: ProfilePrecise, Status: "affected_exact"})
			case "count_none":
				_, err = rawAgg.CountSupplyChainImpactFindings(ctx, AggregateFilter{DetectionProfile: ProfilePrecise})
			}
			if err != nil {
				t.Fatalf("%s: %v", read, err)
			}
		}
		var generic int
		_ = database.QueryRowContext(ctx, `SELECT COALESCE(max(generic_plans), 0) FROM pg_prepared_statements WHERE statement LIKE '%b_candidates AS (%'`).Scan(&generic)
		t.Logf("PLANSTATE %s generic_plans(V statement)=%d", read, generic)
		_ = database.Close()
	}
}
