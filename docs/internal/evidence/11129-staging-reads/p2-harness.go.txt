type p2Clock struct {
	mu       sync.Mutex
	staging  time.Duration
	dirtyLog time.Duration
}

func (c *p2Clock) add(kind int, d time.Duration) {
	c.mu.Lock()
	if kind == 1 {
		c.staging += d
	} else {
		c.dirtyLog += d
	}
	c.mu.Unlock()
}
func (c *p2Clock) take() (time.Duration, time.Duration) {
	c.mu.Lock()
	defer c.mu.Unlock()
	s, l := c.staging, c.dirtyLog
	c.staging, c.dirtyLog = 0, 0
	return s, l
}

var p2StagingStatements = map[string]bool{}
var p2DirtyLogStatements = map[string]bool{}

func p2Kind(q string) int {
	if p2StagingStatements[q] {
		return 1
	}
	if p2DirtyLogStatements[q] {
		return 2
	}
	return 0
}

type p2DB struct {
	inner SQLDB
	clock *p2Clock
}

func (d p2DB) QueryContext(ctx context.Context, q string, args ...any) (db.Rows, error) {
	return d.inner.QueryContext(ctx, q, args...)
}
func (d p2DB) ExecContext(ctx context.Context, q string, args ...any) (sql.Result, error) {
	return d.inner.ExecContext(ctx, q, args...)
}
func (d p2DB) Begin(ctx context.Context) (db.Transaction, error) {
	tx, err := d.inner.Begin(ctx)
	if err != nil {
		return nil, err
	}
	return p2Tx{inner: tx, clock: d.clock}, nil
}

type p2Tx struct {
	inner db.Transaction
	clock *p2Clock
}

// p2Rows drains a staging statement's cursor inside the timer.
func (t p2Tx) QueryContext(ctx context.Context, q string, args ...any) (db.Rows, error) {
	start := time.Now()
	rows, err := t.inner.QueryContext(ctx, q, args...)
	if k := p2Kind(q); k != 0 && err == nil {
		return &p2Rows{Rows: rows, start: start, clock: t.clock, kind: k}, err
	}
	return rows, err
}
func (t p2Tx) ExecContext(ctx context.Context, q string, args ...any) (sql.Result, error) {
	return t.inner.ExecContext(ctx, q, args...)
}
func (t p2Tx) Commit() error   { return t.inner.Commit() }
func (t p2Tx) Rollback() error { return t.inner.Rollback() }

type p2Rows struct {
	db.Rows
	start time.Time
	clock *p2Clock
	kind  int
	done  bool
}

func (r *p2Rows) Close() error {
	err := r.Rows.Close()
	if !r.done {
		r.done = true
		r.clock.add(r.kind, time.Since(r.start))
	}
	return err
}

func p2Summary(d []time.Duration) string {
	s := append([]time.Duration(nil), d...)
	sort.Slice(s, func(i, j int) bool { return s[i] < s[j] })
	var tot time.Duration
	for _, x := range s {
		tot += x
	}
	q := func(p float64) time.Duration { return s[int(p*float64(len(s)-1))] }
	return "n=" + p2itoa(len(s)) + " first=" + d[0].String() + " p50=" + q(0.5).String() + " p95=" + q(0.95).String() + " max=" + s[len(s)-1].String() + " total=" + tot.String()
}

func p2itoa(i int) string { return strconv.Itoa(i) }
func pad6(i int) string {
	s := p2itoa(i)
	for len(s) < 6 {
		s = "0" + s
	}
	return s
}
