"""Applies ruling 11129c's fixed T5-A2 decision rule to run.txt rows."""
import re
import statistics as st
import sys

run = open(sys.argv[1]).read()
old = open(sys.argv[2]).read() if len(sys.argv) > 2 else ""

rows = []
for m in re.finditer(r"T5A2ROW (.*)", run):
    d = dict(kv.split("=", 1) for kv in m.group(1).split())
    rows.append(d)
waits = {}
for m in re.finditer(r"T5A2WAITS cell=(\d+) writers=\d+ index=\w+ (.*)", run):
    waits[int(m.group(1))] = m.group(2)

f = lambda d, k: float(d[k])
pairs = {}
for i in range(0, len(rows), 2):
    a, b = rows[i], rows[i + 1]
    idx, noi = (a, b) if a["index"] == "true" else (b, a)
    assert idx["index"] == "true" and noi["index"] == "false", (a["cell"], b["cell"])
    w = int(idx["writers"])
    marks = f(idx, "marks")
    pairs.setdefault(w, []).append({
        "cells": (noi["cell"], idx["cell"]),
        "tput": 100 * (f(idx, "passes_per_s") / f(noi, "passes_per_s") - 1),
        "cpu": 100 * (f(idx, "cpu_ms_per_pass") / f(noi, "cpu_ms_per_pass") - 1),
        "wal": 100 * (f(idx, "lsn_bytes_per_pass") / f(noi, "lsn_bytes_per_pass") - 1),
        "rec_per_mark": (f(idx, "all_rec_bytes") - f(noi, "all_rec_bytes")) / marks,
        "fpi_bytes_per_mark": (f(idx, "all_fpi_bytes") - f(noi, "all_fpi_bytes")) / marks,
        "fpi_count_per_mark": (f(idx, "wal_fpi") - f(noi, "wal_fpi")) / marks,
        "btree_rec_per_pass": (f(idx, "btree_records") - f(noi, "btree_records")) / f(idx, "passes"),
        "lock_pp": 100 * (f(idx, "lock_share") - f(noi, "lock_share")),
        "noidx_lock": 100 * f(noi, "lock_share"),
        "loads": (noi["load_start"], noi["load_end"], idx["load_start"], idx["load_end"]),
    })

def mean(xs):
    return st.mean(xs) if xs else float("nan")

def se(xs):
    return st.stdev(xs) / len(xs) ** 0.5 if len(xs) > 1 else float("nan")

for w in sorted(pairs):
    print(f"== {w} writers, {len(pairs[w])} pairs")
    for p in pairs[w]:
        print("  cells %s/%s tput %+.2f%% cpu %+.2f%% wal %+.2f%% rec/mark %+.1fB fpiB/mark %+.1f fpi#/mark %+.5f btree_rec/pass %+.1f lock %+.2fpp (noidx %.2f%%) loads %s" % (
            p["cells"][0], p["cells"][1], p["tput"], p["cpu"], p["wal"], p["rec_per_mark"], p["fpi_bytes_per_mark"],
            p["fpi_count_per_mark"], p["btree_rec_per_pass"], p["lock_pp"], p["noidx_lock"], "/".join(p["loads"])))
    for k in ("tput", "cpu", "wal", "rec_per_mark", "fpi_bytes_per_mark", "fpi_count_per_mark", "btree_rec_per_pass", "lock_pp", "noidx_lock"):
        xs = [p[k] for p in pairs[w]]
        print(f"  mean {k} = {mean(xs):+.4f} (SE {se(xs):.4f}, min {min(xs):+.4f}, max {max(xs):+.4f})")

# Old round 8-writer throughput pair deltas for the pooled R1 throughput mean.
old_tput = []
if old:
    r = re.findall(r"T5A writers=8 index=(\w+) .*?passes/s=([0-9.]+)", old)
    for i in range(0, len(r), 2):
        (ia, va), (ib, vb) = r[i], r[i + 1]
        idxv, noiv = (float(va), float(vb)) if ia == "true" else (float(vb), float(va))
        old_tput.append(100 * (idxv / noiv - 1))
print("== decision rule (ruling 11129c)")
p8 = pairs.get(8, [])
pooled = old_tput + [p["tput"] for p in p8]
r1 = {
    "throughput pooled mean >= -5%": (mean(pooled), mean(pooled) >= -5),
    "cpu/pass <= +5%": (mean([p["cpu"] for p in p8]), mean([p["cpu"] for p in p8]) <= 5),
    "wal bytes/pass <= +5%": (mean([p["wal"] for p in p8]), mean([p["wal"] for p in p8]) <= 5),
    "lock+lwlock share rise <= 2pp": (mean([p["lock_pp"] for p in p8]), mean([p["lock_pp"] for p in p8]) <= 2),
}
print(f"  R1 pooled throughput pairs: old {len(old_tput)} new {len(p8)}")
for k, (v, ok) in r1.items():
    print(f"  R1 {k}: {v:+.3f} {'PASS' if ok else 'FAIL'}")
p1, p16 = pairs.get(1, []), pairs.get(16, [])
rec1, rec16 = mean([p["rec_per_mark"] for p in p1]), mean([p["rec_per_mark"] for p in p16])
cpu1, cpu16 = mean([p["cpu"] for p in p1]), mean([p["cpu"] for p in p16])
lock16 = mean([p["lock_pp"] for p in p16])
print(f"  R2 non-FPI record bytes/mark 16w {rec16:+.2f} / 1w {rec1:+.2f} = {rec16 / rec1:.3f}x {'PASS' if rec16 <= 2 * rec1 else 'FAIL'}")
print(f"  R2 cpu overhead 16w {cpu16:+.3f}% / 1w {cpu1:+.3f}% = {cpu16 / cpu1 if cpu1 else float('nan'):.3f}x {'PASS' if cpu16 <= 2 * cpu1 else 'FAIL'}")
print(f"  R2 lock+lwlock share rise at 16w {lock16:+.3f}pp {'PASS' if lock16 <= 2 else 'FAIL'}")
f1 = mean([p["fpi_count_per_mark"] for p in p1])
over = sum(1 for p in p16 if (p["fpi_count_per_mark"] > 2 * f1 and (p["fpi_count_per_mark"] > 0) == (f1 > 0)))
print(f"  FPI count/mark: 1w mean {f1:+.5f}; 16w pairs over 2x with same sign: {over} of {len(p16)} {'RETURN-TO-ARBITER' if over >= 5 else 'ok'}")
for w in sorted(pairs):
    xs = [p["noidx_lock"] for p in pairs[w]]
    spread = max(xs) - min(xs)
    print(f"  no-index lock share spread at {w}w: {spread:.2f}pp {'RETURN-TO-ARBITER' if spread > 2 else 'ok'}")
print("== no-index wait profiles (limiter)")
for d in rows:
    if d["index"] == "false":
        print(f"  cell {d['cell']} w{d['writers']}: active={d['active_samples']} {waits.get(int(d['cell']), '')}")
