  publish-ci-service-mirrors:
    if: >-
      github.event_name == 'workflow_dispatch' && inputs.mode == 'ci-mirrors-publish' &&
      github.repository == 'eshu-hq/eshu' &&
      github.ref == 'refs/heads/main'
    runs-on: ubuntu-latest
    timeout-minutes: 30
    permissions:
      contents: read
      packages: write
    steps:
      - uses: actions/checkout@v5
      - uses: actions/setup-go@v6
        with:
          go-version-file: go/go.mod
          cache: false
      - name: Install pinned crane
        run: GOBIN="${RUNNER_TEMP}" scripts/ci/go-install-retry.sh github.com/google/go-containerregistry/cmd/crane@v0.20.6
      - name: Install ripgrep
        run: scripts/ci/install-apt-packages.sh ripgrep
      - name: Test publisher safety contract
        run: bash scripts/dev/test-publish-ci-image-mirrors.sh
      - name: Log in to GHCR
        uses: docker/login-action@v3
        with:
          registry: ghcr.io
          username: ${{ github.actor }}
          password: ${{ secrets.GITHUB_TOKEN }}
      - name: Copy exact upstream indexes
        env:
          CRANE_BIN: ${{ runner.temp }}/crane
          EXPECTED_REVIEWED_SHA: ${{ inputs.expected_sha }}
        run: bash scripts/dev/publish-ci-image-mirrors.sh publish
